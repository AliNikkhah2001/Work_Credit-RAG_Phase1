\noindent\texttt{stream} defaults to \texttt{true}.

\subsection{Response (\texttt{stream: false})}

\begin{lstlisting}
{
  "id": "chatcmpl-...",
  "object": "chat.completion",
  "model": "ics-helper-agent",
  "choices": [
    {
      "index": 0,
      "message": { "role": "assistant", "content": "..." },
      "finish_reason": "stop"
    }
  ],
  "usage": null,
  "rag": {
    "request_id": "...",
    "citations": [
      {
        "chunk_id": "...",
        "document_id": "...",
        "title": "...",
        "heading": "..."
      }
    ],
    "supplemental": null,
    "follow_ups": ["..."],
    "applied_subspace": "general",
    "sticky_subspace": "general",
    "available_subspaces": null
  },
  "audit": null
}
\end{lstlisting}

\noindent\texttt{finish_reason}: \texttt{stop} $|$ \texttt{clarify} $|$ \texttt{content_filter} $|$ \texttt{length} $|$ \texttt{error}

\subsection{Response (\texttt{stream: true}) --- SSE}

\noindent Media type: \texttt{text/event-stream}

\begin{lstlisting}
data: {"id":"chatcmpl-...","object":"chat.completion.chunk","choices":[{"index":0,"delta":{"role":"assistant"},"finish_reason":null}]}

data: {"id":"chatcmpl-...","object":"chat.completion.chunk","choices":[{"index":0,"delta":{"content":"اعتبارسنجی "},"finish_reason":null}]}

data: {"id":"chatcmpl-...","object":"chat.completion.chunk","choices":[{"index":0,"delta":{"content":"فرآیندی "},"finish_reason":null}]}

data: {"id":"chatcmpl-...","object":"chat.completion.chunk","choices":[{"index":0,"delta":{},"finish_reason":"stop"}],"rag":{"request_id":"...","citations":[],"follow_ups":[]}}

data: [DONE]
\end{lstlisting}

\begin{itemize}
  \item Lines starting with \texttt{:} --- ignore
  \item Append every \texttt{choices[0].delta.content}
  \item Final chunk may include \texttt{rag}
  \item End on \texttt{data: [DONE]}
\end{itemize}

\section{Rating --- \texttt{/rate}}

\noindent Same endpoint. Last user message:

\begin{lstlisting}
/rate <1-5> [comment] [#tag ...]
\end{lstlisting}

\subsection{Request}

\begin{lstlisting}
{
  "model": "ics-helper-agent",
  "messages": [
    { "role": "user", "content": "/rate 5 پاسخ خوب بود #helpful" }
  ],
  "stream": false
}
\end{lstlisting}

\noindent Use the same \texttt{X-Request-ID} as the rated answer.

\subsection{Response}

\begin{lstlisting}
{
  "id": "chatcmpl-...",
  "object": "chat.completion",
  "model": "ics-helper-agent",
  "choices": [
    {
      "index": 0,
      "message": {
        "role": "assistant",
        "content": "Rating saved: ***** (5/5)\n> پاسخ خوب بود  Tags: #helpful"
      },
      "finish_reason": "stop"
    }
  ],
  "rag": { "request_id": "...", "citations": [] },
  "audit": null
}
\end{lstlisting}
