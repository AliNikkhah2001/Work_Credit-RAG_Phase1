_fuse_subspace_rrf() def _fuse_subspace_rrf(filtered, whole, filtered_weight=1.3): """ Reciprocal Rank Fusion with filtered boost score = 1/(60 + rank) * weight """ merged = {} # Filtered leg: weight 1.3x (boosted) for rank, chunk in enumerate(filtered): key = chunk.chunk_id score = filtered_weight / (60 + rank + 1) merged[key] = max(merged.get(key, 0), score) # Whole-KB leg: weight 1.0 for rank, chunk in enumerate(whole): key = chunk.chunk_id score = 1.0 / (60 + rank + 1) merged[key] = max(merged.get(key, 0), score) # Sort by fused score, take top 20 return sorted(merged.values(), key=lambda x: x.fused_score, reverse=True)[:20] ``` --- ## Architecture: Our Fusion vs Papers ```mermaid flowchart TD A[Query] --> B[Filtered Retrieval
