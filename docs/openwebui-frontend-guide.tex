\section{چگونه UI را روی Open WebUI ساختیم}

\begin{enumerate}
  \item پک تم \textenglish{JSON} در \textenglish{\texttt{static/dana/themes/}}
    (مثلاً \textenglish{\texttt{ics-maroon}} سفید/روشن).
  \item \textenglish{\texttt{theme.ts}} پک را می‌خواند، متغیرهای \textenglish{\texttt{--dana-*}}،
    \textenglish{RTL}، نام محصول و \textenglish{\texttt{html.dana-theme}} را ست می‌کند.
  \item \textenglish{\texttt{dana.css}} ظاهر شیشه‌ای و \textenglish{empty-state} را روی برند فعال می‌کند.
  \item \textenglish{\texttt{DanaChatBackground.svelte}} پس‌زمینه موشن/\textenglish{still} را از پک می‌کشد.
  \item ویجت‌های جادویی از \textenglish{\texttt{rag.magic}} (پس از تولید پاسخ)
    یا به‌عنوان fallback از توکن‌های \textenglish{allowlist} در محتوای قدیمی رندر می‌شوند.
    ارکستراتور دیگر \textenglish{\texttt{<<<MAGIC:…>>>}} را داخل بدنه پیام نمی‌نویسد.
  \item بعد از تغییر \textenglish{\texttt{static/dana/}}، به
    \textenglish{\texttt{backend/open\_webui/static/dana/}} هم کپی کنید (یا image را rebuild کنید).
\end{enumerate}

\section{تست طراحی}

\subsection{\textenglish{A} --- Mock OpenAI (فقط UI)}

هر سرور سازگار با \textenglish{OpenAI} که \textenglish{SSE} بدهد:

\begin{english}
\begin{lstlisting}
data: {"choices":[{"delta":{"content":"سلام "}}]}

data: {"choices":[{"delta":{},"finish_reason":"stop"}]}

data: [DONE]
\end{lstlisting}
\end{english}

\textenglish{\texttt{OPENAI\_API\_BASE\_URL}} را به mock بزنید --- برای تم،
\textenglish{RTL}، \textenglish{empty-state}، بدون \textenglish{RAG}.

\subsection{\textenglish{B} --- ارکستراتور واقعی}

\begin{english}
\begin{lstlisting}
curl -s http://127.0.0.1:8100/health
curl -s http://127.0.0.1:8100/v1/models
curl -s http://127.0.0.1:8100/v1/chat/completions \
  -H 'Content-Type: application/json' \
  -H 'Authorization: Bearer sk-local-dev' \
  -H 'X-Request-ID: ui-1' \
  -d '{"model":"ics-helper-agent","messages":[{"role":"user","content":"سلام"}],"stream":false}'
\end{lstlisting}
\end{english}

سپس: \url{http://127.0.0.1:13000} و گالری تم.

\subsection{بازسازی محلی}

\begin{itemize}
  \item فقط \textenglish{CSS/JSON/HTML}: کپی به کانتینر
    \textenglish{\texttt{/app/backend/open\_webui/static/dana/}}
  \item تغییر \textenglish{Svelte}: \textenglish{\texttt{npm run build}} سپس recreate سرویس
    \textenglish{\texttt{webui}}
  \item \textenglish{Hard-refresh}؛ در صورت نیاز
    \textenglish{\texttt{localStorage.dana.theme\_id}} را پاک کنید
\end{itemize}

\section{چک‌لیست سرعت}

\begin{itemize}
  \item اول \textenglish{\texttt{ics-maroon.json}} و \textenglish{\texttt{dana.css}}
  \item گالری قبل از rebuild کامل
  \item sync کردن \textenglish{\texttt{backend/open\_webui/static/dana/}}
  \item \textenglish{magic} فقط از \textenglish{allowlist}
  \item \textenglish{PR} روی \textenglish{WebUI \texttt{develop}}؛ umbrella فقط pin را جلو می‌برد
\end{itemize}
